% ---------- Robust project figure macro ----------
% Usage: \projectfigure[width]{filename}{caption}{label}
% It compiles even if a collaborator has not yet pulled the figure.
\newcommand{\projectfigure}[4][0.94\linewidth]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{Figures/#2}{%
      \includegraphics[width=#1]{#2}%
    }{%
      \fbox{\parbox[c][4.0cm][c]{0.82\linewidth}{\centering
      Missing project figure: \\ \texttt{Figures/#2}}}%
    }
    \caption{#3}
    \label{#4}
  \end{figure}%
}
